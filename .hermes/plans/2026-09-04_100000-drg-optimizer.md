```ts
// Port 1:1 จาก DRGSeeker web/assets/app.js :: buildPayload + validateCaseInput ranges
export const DRG_API_BASE = 'https://had-api.moph.go.th/cmi';
export const DRG_VERSION = '6';
export const MAX_SDX = 12;
export interface DrgCaseInput { hcode:string; sex:1|2; age:number; ageDay:number; weight:number;
  losDay:number; losHour:number; dcCode:string; pdx:string; sdx:string[]; proc:string[]; }
export function buildDrgPayload(v: DrgCaseInput) {
  if (!Array.isArray(v.sdx) || !Array.isArray(v.proc)) throw new Error('ข้อมูลเคสไม่ครบ');
  if (!/^\d{5}$/.test(v.hcode) || !/^[A-Z0-9]+$/.test(v.pdx)) throw new Error('รหัส HCode หรือ PDx ไม่ถูกต้อง');
  if (![v.age, v.ageDay, v.losDay, v.losHour, v.weight].every(Number.isFinite)) throw new Error('ค่าตัวเลขของเคสไม่ถูกต้อง');
  if (!Number.isInteger(v.age) || v.age<0 || v.age>120 || !Number.isInteger(v.ageDay) || v.ageDay<0 || v.ageDay>364
    || !Number.isInteger(v.losDay) || v.losDay<0 || v.losDay>9999 || !Number.isInteger(v.losHour) || v.losHour<0 || v.losHour>23
    || v.weight<0 || v.weight>300) throw new Error('ค่าตัวเลขของเคสอยู่นอกช่วงที่รองรับ');
  if (v.sdx.length > MAX_SDX) throw new Error('SDx เกิน 12 รายการ');
  const dc = /^\d{2}$/.test(v.dcCode) ? v.dcCode : '11';
  return { version: DRG_VERSION, data: [{ hcode:v.hcode, hn:'', an:'1', sex:v.sex, age:v.age,
    age_day:v.ageDay, los_day:v.losDay, los_hour:v.losHour, weight:v.weight,
    dischs:dc.charAt(0), discht:dc.charAt(1)||'1', pdx:v.pdx, sdx:[...v.sdx], proc:[...v.proc] }] };
}
export async function calculateDrg(payload: unknown, signal?: AbortSignal) {
  // ยิงตรงเท่านั้น — ห้ามผ่าน CORS proxy (กติกาเดียวกับ DRGSeeker)
  const res = await fetch(`${DRG_API_BASE}/drg/calculate`, { method:'POST',
    headers:{'Content-Type':'application/json'}, body: JSON.stringify(payload), signal });
  const json = await res.json();
  if (Number(json?.status) !== 200) throw new Error(`Grouper ไม่ตอบกลับผลลัพธ์ (status ${json?.status})`);
  const r = Array.isArray(json.data) ? json.data[0] : null;
  if (!r || r.drg == null || String(r.drg).trim() === '') throw new Error('Grouper ไม่ส่งผลลัพธ์ DRG กลับมา');
  return json as { status:number; data:[{ drg:string; mdc?:string; rw?:number; adjrw?:number; wtlos?:number; ot?:number; err?:string; warn?:string }]; tgrp?: unknown };
}
```

```ts
// shape ลอกจาก cmi-dashboard/src/types/cmi.ts (เฉพาะฟิลด์ที่ใช้)
export interface CmiCaseRow { pdx:string|null; sdx1:string|null; sdx2:string|null; sdx3:string|null; sdx4:string|null;
  extCause:string|null; proc1:string|null; proc2:string|null; proc3:string|null;
  sex:string|null; age:number|null; los:number|null; dchtype:string|null; dchstts:string|null; }
const clean = (s:string|null) => (s||'').trim().toUpperCase().replace(/[^A-Z0-9]/g,'');
export function cmiCaseToDrgInput(row: CmiCaseRow, opts:{ hcode:string; dcCodeFallback?:string }): DrgCaseInput {
  const pdx = clean(row.pdx);
  if (!pdx) throw new Error('เคสนี้ยังไม่ลง PDx — ต้องมี PDx ก่อนเรียก Grouper');
  const sdx = [row.sdx1,row.sdx2,row.sdx3,row.sdx4].map(clean).filter(c=>c && c!==pdx);
  const proc = [row.proc1,row.proc2,row.proc3].map(clean).filter(Boolean);
  const sex: 1|2 = /หญิง|2|F/i.test(row.sex||'') ? 2 : 1;
  const losDay = Number.isInteger(row.los) && (row.los as number) >= 0 ? (row.los as number) : 1;
  const dc = `${(row.dchtype||'').trim().charAt(0)||'1'}${(row.dchstts||'').trim().charAt(0)||'1'}`;
  return { hcode: opts.hcode, sex, age: row.age ?? 0, ageDay: 0, weight: 0,
    losDay, losHour: 0, dcCode: /^\d{2}$/.test(dc) ? dc : (opts.dcCodeFallback ?? '11'), pdx, sdx, proc };
}
```

```ts
export interface UsageLite { hosGuid:string|null; itemName:string|null; needOrderReason:string|null;
  prescReason:string|null; prescReason2:string|null; prescReason3:string|null; prescReason4:string|null; prescReason5:string|null; icode:string|null; }
export interface DxCandidate { code:string; source:'presc_reason'|'need_order_reason'|'coder_manual'; evidence:string[]; }
const CODE_RE = /\b[A-Z][0-9]{2}(?:\.[0-9A-Z]{1,4})?\b/g; // ICD-10 หลวม ๆ แล้วค่อย clean
const clean = (s:string) => s.trim().toUpperCase().replace(/\./g,'').replace(/[^A-Z0-9]/g,'');
export function extractCandidates(items: UsageLite[]): DxCandidate[] {
  const map = new Map<string, DxCandidate>();
  for (const it of items) {
    const fields: Array<[string,string|null]> = [['presc_reason',it.prescReason],['presc_reason',it.prescReason2],
      ['presc_reason',it.prescReason3],['presc_reason',it.prescReason4],['presc_reason',it.prescReason5],
      ['need_order_reason',it.needOrderReason]];
    for (const [src, raw] of fields) {
      if (!raw) continue;
      for (const m of String(raw).toUpperCase().match(CODE_RE) ?? []) {
        const c = clean(m);
        if (c.length < 3) continue;
        const cur = map.get(c) ?? { code:c, source: src as DxCandidate['source'], evidence:[] };
        cur.evidence.push(`${src}:${String(raw).slice(0,80)}${it.hosGuid?` [${it.hosGuid}]`:''}`);
        map.set(c, cur);
      }
    }
  }
  return [...map.values()];
}
```

```ts
import { buildDrgPayload, calculateDrg, type DrgCaseInput } from '@/drg/grouperClient';
export interface Suggestion { kind:'add_sdx'|'swap_pdx'|'add_proc'; pdx:string; sdx:string[]; proc:string[];
  drg:string; adjrw:number|null; delta:number|null; reason:string; warning?:string; }
export async function suggestHigherDrg(base: DrgCaseInput, candidates: string[], opts:{ concurrency?:number; signal?:AbortSignal } = {}): Promise<{ baseline:{drg:string;adjrw:number|null}; suggestions:Suggestion[] }> {
  const bJson = await calculateDrg(buildDrgPayload(base), opts.signal);
  const b = bJson.data[0];
  const baseAdj = typeof b.adjrw === 'number' ? b.adjrw : null;
  const out: Suggestion[] = [];
  // v1: ลองทีละตัวเท่านั้น (add_sdx ก่อน, swap_pdx ทีหลัง) — ห้าม combo ยกเว้น coder กดเอง
  const queue: Array<{kind:Suggestion['kind']; pdx:string; sdx:string[]; proc:string[]; reason:string}> = [];
  for (const c of [...new Set(candidates.map(s=>s.trim().toUpperCase()).filter(s=>s && s!==base.pdx && !base.sdx.includes(s)))].slice(0, 20)) {
    if (base.sdx.length < 12) queue.push({ kind:'add_sdx', pdx:base.pdx, sdx:[...base.sdx, c], proc:base.proc, reason:`พบ ${c} ในหลักฐาน usage (presc/need_order_reason)` });
    queue.push({ kind:'swap_pdx', pdx:c, sdx:[base.pdx, ...base.sdx.filter(x=>x!==c)].slice(0,12), proc:base.proc, reason:`ลอง ${c} เป็น PDx (เดิม ${base.pdx} ลงเป็น SDx)` });
  }
  for (const q of queue) { // sequential v1 กัน rate-limit; ทำ concurrency ทีหลังถ้าช้า
    try {
      const j = await calculateDrg(buildDrgPayload({ ...base, pdx:q.pdx, sdx:q.sdx, proc:q.proc }), opts.signal);
      const r = j.data[0];
      const a = typeof r.adjrw === 'number' ? r.adjrw : null;
      out.push({ ...q, drg: String(r.drg ?? ''), adjrw: a,
        delta: (a != null && baseAdj != null) ? +(a - baseAdj).toFixed(4) : null,
        reason: q.reason, warning: r.err || r.warn ? `grouper: ${r.err||''} ${r.warn||''}`.trim() : undefined });
    } catch { /* ข้ามตัวที่ grouper reject แล้วไปต่อ */ }
  }
  return { baseline:{ drg:String(b.drg), adjrw: baseAdj },
    suggestions: out.filter(s=>s.delta != null && s.delta > 0).sort((x,y)=>(y.delta??0)-(x.delta??0)) };
}
```

## 6. Risks, tradeoffs, and open questions

R1 (clinical safety — ใหญ่สุด): suggest ผิด = upcoding/tุจริตโดยไม่ตั้งใจ → mitigation: ทุก suggestion ต้องมี evidence จาก usage + ป้าย coder-review + ไม่ auto-write กลับ DB เด็ดขาด (read-only ทั้งระบบ)
R2 (ข้อมูลไม่ครบ): CaseDetail มีแค่ sdx1–4/proc1–3, ไม่มี weight/ageDay/losHour/baseRate → mitigation: default 0 + แสดงชัด + ให้ coder แก้ก่อนคำนวณ; tradeoff คือ baseline อาจต่างจาก grouper จริงของ coder ที่กรอกน้ำหนัก
R3 (map DC): `dchtype/dchstts → dischs/discht` ยังไม่มีตารางทางการ → ใช้ fallback '11' ชั่วคราว (tradeoff: DRG บางตัวขึ้นกับ discharge status)
R4 (network): API จำกัด IP ไทย + ไม่มี SLA → mitigation: queue sequential + retry 1 ครั้งเฉพาะ network error (ไม่ retry 4xx), แจ้งเตือนเมื่อ status ≠ 200
R5 (scope creep): อย่าเพิ่ม LLM/auto-coding ใน v1 — YAGNI; v2 ค่อยทำ combo-search/comorbidity-check

Q1 (ถาม coder/พี่พลอย): ตาราง map `dchtype(IPT)+dchstts → dcCode 2 หลัก (/libs/ipd-result)` ของ รพ. คืออะไร? ขอ 5–10 คู่ตัวอย่าง
